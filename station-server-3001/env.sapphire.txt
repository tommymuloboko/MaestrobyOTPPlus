# ===========================================================================

# --- Server ---
PORT=3001
NODE_ENV=production
DEPLOYMENT_MODE=local
SERVER_MODE=local
LOG_LEVEL=info

# --- Station identity (MAESTRO CENTRAL) ---------------------------------------
# These MUST match Sapphire's row in maestro_central (companies + station_registry)
# AND the identity rows seeded in this local DB. Mismatched identity breaks RFID
# auth (see the setup guide — "identity placeholder trap"). Get these UUIDs from
# cloud/3-sapphire-maestro-central.sql when you run it.
COMPANY_ID=SET_SAPPHIRE_CENTRAL_COMPANY_ID
STATION_ID=SET_SAPPHIRE_CENTRAL_STATION_ID
OFFLINE_COMPANY_ID=SET_SAPPHIRE_CENTRAL_COMPANY_ID
OFFLINE_STATION_ID=SET_SAPPHIRE_CENTRAL_STATION_ID
OFFLINE_SCOPE_MODE=station

# --- Local database (MAESTRO-LITE, this PC) -----------------------------------
# DB_MODE=local → endpoints read/write the LOCAL Postgres; sync pushes to central.
# LOCAL Postgres is on 5432 (the OTP server's cloud tunnel is 5433 — different DB).
DB_MODE=local
LOCAL_DB_HOST=localhost
LOCAL_DB_PORT=5432
LOCAL_DB_NAME=maestro_sapphire
LOCAL_DB_USER=postgres
LOCAL_DB_PASSWORD=SET_LOCAL_PG_PASSWORD
LOCAL_DB_POOL_MIN=2
LOCAL_DB_POOL_MAX=10

# --- Sync to Maestro Central (HQ dashboard + price management) ----------------
# Only needed if Sapphire goes on the HQ dashboard. Leave AUTO_PULL off until the
# maestro_central registration (cloud/3-...) is done and the inbound tunnel is up.
AUTO_PULL_ON_STARTUP=false
SYNC_INTERVAL_MS=120000
SYNC_BATCH_SIZE=500
SYNC_MAX_RETRIES=5

# --- PTS-2 forecourt controller (Sapphire's controller on the depot LAN) ------
PTS_POLLING_ENABLED=true
PTS_URL=https://192.168.1.XXX/jsonPTS
PTS_USERNAME=admin
PTS_PASSWORD=SET_PTS_PASSWORD
PTS_TIMEOUT=30000
PTS_POLLING_INTERVAL=2000

# --- RFID attendant readers ---------------------------------------------------
# The OTP server (:3005) reads badges from THIS server. Set RFID_ENABLED=false
# only if the depot has no pump-side tag readers.
RFID_ENABLED=true
RFID_POLL_INTERVAL_MS=1000

# --- Auth ---------------------------------------------------------------------
# SERVICE_TOKEN must EQUAL the OTP server's STATION_SERVICE_TOKEN, and JWT_SECRET/
# SERVICE_TOKEN are what the tablet + OTP server use to reach this server.
# Generate fresh 32-byte hex values; do NOT reuse another station's.
JWT_SECRET=SET_JWT_SECRET
SERVICE_TOKEN=SET_SERVICE_TOKEN
JWT_EXPIRY=12h
BCRYPT_ROUNDS=10
